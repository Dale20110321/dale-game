// 临时回归自检（注入 index.html 的副本，结果写入 #ERRDUMP 供 --dump-dom 抓取）
(function(){
  const R=[],errs=[];
  const say=s=>R.push(s);
  const oldErr=console.error;console.error=(...a)=>{errs.push("console:"+a.join(" ").slice(0,300));oldErr.apply(console,a);};
  addEventListener("error",e=>errs.push("win:"+e.message));
  window.requestAnimationFrame=fn=>{return 1;};        // 接管 rAF：手动驱动，避免无头下不跑帧
  let T=0;
  const step=n=>{for(let i=0;i<n;i++){T+=16.7;try{window.loop(T);}catch(e){errs.push("loop:"+e.message);return;}}};
  const midX=()=>((bike.rear.x+bike.front.x)/2);

  function drive(maxSec,stop){
    key.right=true;key.left=false;
    let cr=0,was=false,minF=9,stuck=0,lx=midX();
    for(let i=0;i<Math.round(maxSec*60);i++){
      step(1);
      if(crashed&&!was){cr++;was=true;}
      if(!crashed)was=false;
      minF=Math.min(minF,fuel/fuelMax);
      const x=midX();
      if(x-lx<0.25)stuck++;else stuck=0;
      lx=x;
      if(stuck>300)return{cr,minF,stuck:true};
      if(stop&&stop())return{cr,minF,stuck:false};
    }
    key.right=false;
    return{cr,minF,stuck:false};
  }

  // A. 20 关自动通关回归
  say("== A. 20关通关回归 ==");
  let aFail=0,aCrash=0,aStuck=0;
  for(let lv=0;lv<20;lv++){
    let line="L"+(lv+1)+" ";
    try{
      window.startGame("level",lv);T+=1000;step(3);
      const len=LEVELS[lv].len,t0=T;
      const r=drive(Math.max(50,len/165),()=>clearing===true);
      const fin=midX(),secs=((T-t0)/1000).toFixed(1);
      const passed=fin>=len-40;
      const s=stars[lv]||0;
      line+=(passed?"✅":"❌")+" d="+Math.round(fin)+"/"+len+" t="+secs+"s ★"+s+" 摔"+r.cr+" 油"+Math.round(r.minF*100)+"%";
      if(!passed){aFail++;line+=(r.stuck?" 卡死":" 超时");}
      aCrash+=r.cr;if(r.stuck)aStuck++;
    }catch(e){aFail++;line+="❌ 异常:"+e.message;}
    say(line);
  }
  say("A汇总: 失败"+aFail+" 摔车合计"+aCrash+" 卡死"+aStuck);

  // B. 陡坡重生起步（第20关 + 第16关采样，走真实 respawn 路径）
  say("== B. 重生起步（真实 respawn） ==");
  for(const LV of [19,15,9]){
    try{
      window.startGame("level",LV);step(3);
      let bad=0,tested=0,worst="";
      for(let x=600;x<LEVELS[LV].len-300;x+=400){
        const gi=groundInfo(x);if(!isFinite(gi.y))continue;
        lastSafeX=x;
        window.respawn();step(2);
        const x0=bike.rear.x;let mx=x0;
        key.right=true;
        for(let i=0;i<240;i++){step(1);if(bike.rear.x>mx)mx=bike.rear.x;}
        key.right=false;
        tested++;
        if(mx-x0<60){bad++;if(!worst)worst="首个: 原x="+Math.round(x)+" 坡度"+Math.round(Math.atan(Math.abs(gi.m))*57.3)+"° 实际只走"+Math.round(mx-x0)+"px";}
      }
      say("L"+(LV+1)+" 采样"+tested+" 起步失败"+bad+(worst?" ❌ "+worst:" ✅"));
    }catch(e){say("L"+(LV+1)+" 异常:"+e.message);}
  }

  // C. 关卡参数分层核查
  say("== C. 分层参数 ==");
  for(const lv of [0,3,7,10,14,19]){
    window.startGame("level",lv);
    say("L"+(lv+1)+" 坡度"+LEVELS[lv].maxSlope.toFixed(1)+"° ramp="+LEVELS[lv].ramp.toFixed(2)+" 三星基准"+LEVELS[lv].den3.toFixed(0)+"px/s 油罐"+canisters.length+" 加速带"+boosts.length);
  }

  // D. 成就 / 面板 / 模式冒烟
  say("== D. 系统冒烟 ==");
  try{
    window.renderAchPanel();
    say("成就项="+document.querySelectorAll(".achItm").length+"/"+ACHS.length+" 标题="+(document.querySelector(".modeTitle")?document.querySelector(".modeTitle").textContent:"无"));
    window.checkAch("rich");say("checkAch 后 已达成="+achGot.length);
    window.backToMenu();
  }catch(e){say("成就异常:"+e.message);}
  try{
    window.startGame("free");step(180);
    say("自由模式 3s: x="+Math.round(midX())+" 主题="+curTheme+" 错误="+errs.length);
  }catch(e){say("自由模式异常:"+e.message);}
  try{
    window.startGame("race",5);T+=1000;step(240);
    say("比赛模式: 玩家x="+Math.round(midX())+" AIx="+Math.round(raceAI.x)+" 存活="+(state==="play"));
  }catch(e){say("比赛异常:"+e.message);}
  try{
    window.startGame("level",19);step(60);
    say("HUD速度文本="+HUD.speed.textContent+" 关卡文本="+HUD.lvl.textContent);
  }catch(e){say("HUD异常:"+e.message);}

  say("== 错误汇总: "+errs.length+" ==");
  errs.slice(0,6).forEach(e=>say("  ! "+e));
  const d=document.createElement("div");d.id="ERRDUMP";d.textContent="\n[[VERIFY]]\n"+R.join("\n")+"\n[[END]]\n";
  document.body.appendChild(d);
})();
